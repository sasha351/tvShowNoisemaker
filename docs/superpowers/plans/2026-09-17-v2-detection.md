# v2 Detection Fixes (Phase 2) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make intro/credits detection trustworthy: the season decides what the shared intro and credits are (consensus, not longest match), episodes that do not match are flagged for review instead of silently filled, cuts snap toward the safe side and keep a margin, and every episode carries a confidence verdict.

**Architecture:** `detector._detect_by_chromaprint` collects every pair match, clusters them by length and publishes a `SeasonConsensus` (side channel next to the existing fingerprint stats). `pipeline._resolve_episode` turns fingerprint results plus consensus into boundaries, `confidence` and human-readable `flags`; silence detection is only a fallback when fingerprinting did not run. `_snap_to_silence` prefers the side of a boundary where moving the cut keeps content, then `cut_margin` (Config, default 0.5 s) is applied outward. `tools/bench_detection.py` against `tools/ground_truth.json` is the acceptance gate.

**Tech Stack:** Python 3.10+, numpy, ffmpeg/fpcalc via subprocess, pytest + unittest.mock.

Spec: `docs/superpowers/specs/2026-09-17-v2-design.md` (Phase 2). Prior plan (Phases 0–1, shipped): `docs/superpowers/plans/2026-09-17-v2-pipeline.md`.

```python
# ── Season consensus ────────────────────────────────────────────────────────

_CONSENSUS_TOLERANCE = 3.0  # seconds; pair matches within this of a cluster centre vote together

PairMatch = Tuple[str, Tuple[float, float], str, Tuple[float, float]]


@dataclass
class MatchInfo:
    """How strongly one episode supports the season consensus."""

    intro_support: int = 0            # pair matches inside the winning intro cluster
    outro_support: int = 0
    intro_len: Optional[float] = None  # this episode's raw (pre-snap) matched intro length
    outro_len: Optional[float] = None
    intro_outliers: List[float] = field(default_factory=list)  # lengths of ignored intro matches (recaps…)
    outro_outliers: List[float] = field(default_factory=list)


@dataclass
class SeasonConsensus:
    """What the season agrees on: shared intro/credits length and who supports it."""

    intro_len: Optional[float] = None  # median length of the winning intro cluster; None = no shared intro
    outro_len: Optional[float] = None  # same for credits, measured from the end of the file
    intro_episodes: int = 0            # episodes covered by the winning cluster
    outro_episodes: int = 0
    episodes: Dict[str, MatchInfo] = field(default_factory=dict)


# ponytail: same module-level side channel as the fingerprint stats; the
# public detect_by_fingerprinting() return type stays a plain dict.
_last_consensus = SeasonConsensus()


def _set_last_consensus(consensus: SeasonConsensus) -> None:
    global _last_consensus
    _last_consensus = consensus


def get_last_consensus() -> SeasonConsensus:
    """Consensus published by the most recent Chromaprint pass."""
    return _last_consensus


@dataclass
class _Consensus:
    segments: Dict[str, Tuple[float, float]] = field(default_factory=dict)  # per-episode segment in the winning cluster
    support: Dict[str, int] = field(default_factory=dict)                   # cluster matches per episode
    outliers: Dict[str, List[float]] = field(default_factory=dict)          # ignored match lengths per episode
    median_len: Optional[float] = None                                       # None = no consensus


def _consensus_segments(
    matches: List[PairMatch], n_episodes: int, tolerance: float = _CONSENSUS_TOLERANCE,
) -> _Consensus:
    """Pick the shared-segment length most episodes agree on, and each episode's segment in it.

    Every pair match votes once per side with its length.  Candidate clusters
    are centred on each distinct length; a vote joins a cluster when its length
    is within *tolerance* of the centre.  The winning cluster covers the most
    episodes (ties → longer) and must cover ``max(2, ceil(n_episodes / 3))`` of
    them, otherwise the season has no consensus.  An episode's segment is the
    median of its cluster votes; votes outside the cluster (recaps, two-parters,
    shared cues) are returned as outlier lengths so callers can flag them.
    """
    out = _Consensus()
    votes = []  # (length, episode, segment)
    for a, seg_a, b, seg_b in matches:
        votes.append((seg_a[1] - seg_a[0], a, seg_a))
        votes.append((seg_b[1] - seg_b[0], b, seg_b))
    if not votes:
        return out

    best_center, best_cover = 0.0, 0
    for center in sorted({v[0] for v in votes}, reverse=True):  # longer first so ties keep the longer
        cover = len({ep for length, ep, _ in votes if abs(length - center) <= tolerance})
        if cover > best_cover:
            best_center, best_cover = center, cover
    if best_cover < max(2, math.ceil(n_episodes / 3)):
        for length, ep, _ in votes:
            out.outliers.setdefault(ep, []).append(round(length, 1))
        return out

    by_ep: Dict[str, List[Tuple[float, float]]] = {}
    for length, ep, seg in votes:
        if abs(length - best_center) <= tolerance:
            by_ep.setdefault(ep, []).append(seg)
        else:
            out.outliers.setdefault(ep, []).append(round(length, 1))
    out.median_len = float(np.median([length for length, _, _ in votes if abs(length - best_center) <= tolerance]))
    for ep, segs in by_ep.items():
        out.segments[ep] = (float(np.median([s[0] for s in segs])), float(np.median([s[1] for s in segs])))
        out.support[ep] = len(segs)
    return out
```

```python
def _pairwise_matches(
    fps: Dict[str, np.ndarray],
    paths: List[str],
    min_dur: float,
    max_dur: float,
    max_workers: int,
) -> List[PairMatch]:
    """Compare every episode pair; return every match as ``(a, (a0, a1), b, (b0, b1))``.

    Deterministic: pairs are compared in submission order.
    """
    pairs = [(a, b) for i, a in enumerate(paths) for b in paths[i + 1:]]

    def _compare(pair: Tuple[str, str]):
        a, b = pair
        try:
            return pair, _compare_episodes_chromaprint(
                fps[a], fps[b], min_dur=min_dur, max_dur=max_dur,
            )
        except Exception as exc:  # pylint: disable=broad-except
            logger.debug("Chromaprint compare failed for %s vs %s: %s", a, b, exc)
            return pair, (None, None)

    matches: List[PairMatch] = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as pool:
        for (a, b), (lr, rr) in pool.map(_compare, pairs):
            if lr is not None and rr is not None:
                matches.append((a, lr, b, rr))
    return matches
```

```python
        intro_len=intro.median_len, outro_len=outro.median_len,
        intro_episodes=len(intro.segments), outro_episodes=len(outro.segments),
    )

    # ── Assemble per-episode results: None where this episode did not match ──
    results: Dict[str, Tuple[Optional[float], Optional[float], Optional[float]]] = {}
    for fp in valid_paths:
        dur = durations[fp]
        is_: Optional[float] = None
        ie: Optional[float] = None
        os_: Optional[float] = None
        if fp in intro.segments:
            is_, ie = intro.segments[fp]
        if fp in outro.segments:
            # Convert reversed-fingerprint coordinates back to absolute time.
            os_ = dur - outro.segments[fp][1]
            if ie is not None and os_ < ie:
                os_ = ie
        results[fp] = (is_, ie, os_)
        consensus.episodes[fp] = MatchInfo(
            intro_support=intro.support.get(fp, 0),
            outro_support=outro.support.get(fp, 0),
            intro_len=(ie - is_) if ie is not None and is_ is not None else None,
            outro_len=(dur - os_) if os_ is not None else None,
            intro_outliers=intro.outliers.get(fp, []),
            outro_outliers=outro.outliers.get(fp, []),
        )
        logger.debug(
            "Chromaprint result for %s: intro=%.1fs–%.1fs, outro=%.1fs",
            Path(fp).name,
            is_ if is_ is not None else 0.0,
            ie if ie is not None else 0.0,
            os_ if os_ is not None else dur,
        )
    _set_last_consensus(consensus)
```

| fingerprint value | season consensus | result |
|---|---|---|
| `None` | none for the season | intro: `intro_start=None, intro_end=0.0`; outro: `outro_start=duration`. Flag `"no shared intro found in this season"` / `"no shared credits found in this season"`. Confidence unaffected. |

| value, support ≥ 2, raw length within 3 s of season | exists | boundary used, confidence unaffected |
| value, support < 2 | exists | boundary used, `review`, flag `"intro matched 1 episode only"` / `"credits matched 1 episode only"` |
| value, raw length off by > 3 s | exists | boundary used, `review`, flag `"intro length 48 s vs season 31 s"` / `"credits length 21 s vs season 39 s"` |

Outlier matches always add an informational flag `"ignored 1 other shared segment(s) near the start (45 s)"` / `"… near the end (…)"` without changing confidence. Episodes not in `fingerprint_results` (fingerprinting off/failed, duration unknown) go through `resolve_boundaries` as before: `source="config"` + `high` when timestamps or user durations apply, otherwise `source="silence"` + `review` + flag `"silence detection only (no fingerprint match)"`. Config timestamps for an episode always win (`source="config"`, `high`), even when fingerprinted. The skip rules are unchanged: nothing detected at all → `"No intro/outro boundaries detected (skipped to avoid full-file conversion)"`; `outro_start <= intro_end` → `"Invalid boundaries: …"`.

```python
def _resolve_episode(
    fp: str,
    duration: Optional[float],
    fingerprint_results: Dict[str, Tuple[Optional[float], Optional[float], Optional[float]]],
    cfg: Config,
    consensus: SeasonConsensus,
) -> EpisodePlan:
    """Resolve one episode's boundaries, confidence and flags."""
    ep = EpisodePlan(path=fp, duration=duration)
    try:
        ts = cfg.timestamps.get(ep.basename, {})
        has_timestamp_override = ts.get("intro_end") is not None or ts.get("outro_start") is not None
        has_user_duration_override = cfg.intro_duration is not None or cfg.outro_duration is not None

        if fp in fingerprint_results and not has_timestamp_override:
            _resolve_from_consensus(ep, fingerprint_results[fp], consensus.episodes.get(fp, MatchInfo()), consensus)
        else:
            ep.intro_start, ep.intro_end, ep.outro_start = resolve_boundaries(
                fp, cfg=cfg, fingerprint_results=fingerprint_results, duration=duration,
            )
            if has_timestamp_override or has_user_duration_override:
                ep.source = "config"
            else:
                ep.source = "silence"
                ep.confidence = "review"
                ep.flags.append("silence detection only (no fingerprint match)")

        # Nothing to cut at either end and no explicit user choice: skip rather than convert the whole file.
        nothing_detected = (
            duration is not None
            and ep.intro_end is not None and abs(ep.intro_end) < 1e-6
            and ep.outro_start is not None and abs(ep.outro_start - duration) < 1e-3
        )
        if nothing_detected and ep.source != "config":
            ep.confidence = "skip"
            # Skip reasons go first: encode() and the log report flags[0].
            ep.flags.insert(0, "No intro/outro boundaries detected (skipped to avoid full-file conversion)")
            return ep

        # A None boundary just leaves `encodable` False; only compare when both are known.
        if ep.intro_end is not None and ep.outro_start is not None and ep.outro_start <= ep.intro_end:
            ep.confidence = "skip"
            ep.flags.insert(
                0, f"Invalid boundaries: intro_end={ep.intro_end:.2f}s >= outro_start={ep.outro_start:.2f}s",
            )
    except Exception as exc:  # pylint: disable=broad-except
        ep.confidence = "skip"
        ep.flags = [f"Boundary detection error: {exc}"]
    return ep


def _resolve_from_consensus(
    ep: EpisodePlan,
    boundaries: Tuple[Optional[float], Optional[float], Optional[float]],
    info: MatchInfo,
    consensus: SeasonConsensus,
) -> None:
    """Apply the decision table: use matched boundaries, flag weak ones, never fill silently."""
    intro_start, intro_end, outro_start = boundaries
    ep.source = "fingerprint"

    if intro_end is None and consensus.intro_len is None:
        ep.intro_start, ep.intro_end = None, 0.0
        ep.flags.append("no shared intro found in this season")
    elif intro_end is None:
        ep.intro_start, ep.intro_end = None, None
        ep.confidence = "review"
        ep.flags.append(f"no intro match (season intro {consensus.intro_len:.0f} s)")
    else:
        ep.intro_start, ep.intro_end = intro_start, intro_end
        if info.intro_support < 2:
            ep.confidence = "review"
            ep.flags.append("intro matched 1 episode only")
        if (
            info.intro_len is not None and consensus.intro_len is not None
            and abs(info.intro_len - consensus.intro_len) > _CONSENSUS_LEN_TOLERANCE
        ):
            ep.confidence = "review"
            ep.flags.append(f"intro length {info.intro_len:.0f} s vs season {consensus.intro_len:.0f} s")
    if info.intro_outliers:
        ep.flags.append(
            f"ignored {len(info.intro_outliers)} other shared segment(s) near the start "
            f"({', '.join(f'{x:.0f} s' for x in info.intro_outliers)})"
        )

    if outro_start is None and consensus.outro_len is None:
        ep.outro_start = ep.duration
        ep.flags.append("no shared credits found in this season")
    elif outro_start is None:
        ep.outro_start = None
        ep.confidence = "review"
        ep.flags.append(f"no credits match (season credits {consensus.outro_len:.0f} s)")
    else:
        ep.outro_start = outro_start
        if info.outro_support < 2:
            ep.confidence = "review"
            ep.flags.append("credits matched 1 episode only")
        if (
            info.outro_len is not None and consensus.outro_len is not None
            and abs(info.outro_len - consensus.outro_len) > _CONSENSUS_LEN_TOLERANCE
        ):
            ep.confidence = "review"
            ep.flags.append(f"credits length {info.outro_len:.0f} s vs season {consensus.outro_len:.0f} s")
    if info.outro_outliers:
        ep.flags.append(
            f"ignored {len(info.outro_outliers)} other shared segment(s) near the end "
            f"({', '.join(f'{x:.0f} s' for x in info.outro_outliers)})"
        )
```

```python
def _snap_to_silence(filepath: str, t: float, duration: float, safe: str = "before") -> float:
    """Snap boundary *t* to the start of a nearby silence, preferring the safe side.

    Searches [t − _REFINE_WINDOW_BEFORE, t + _REFINE_WINDOW_AFTER] like
    intro-skipper.  *safe* names the side where moving the cut can only keep
    more content: ``"before"`` for intro_end (earlier keeps a little theme),
    ``"after"`` for intro_start and outro_start (later keeps a little cold open
    or episode).  Silences on the safe side win, closest first; the unsafe side
    is used only when the safe side has none, and then only the silence nearest
    *t* — never one past spoken words.  Returns *t* unchanged when no silence is
    found or *t* is at the file edges.
    """
    if t <= _REFINE_WINDOW_BEFORE or t >= duration - _REFINE_WINDOW_AFTER:
        return t

    result = _run_silencedetect(
        filepath,
        _REFINE_NOISE_DB,
        _REFINE_MIN_SILENCE,
        seek=t - _REFINE_WINDOW_BEFORE,
        limit=_REFINE_WINDOW_BEFORE + _REFINE_WINDOW_AFTER,
    )
    if result is None:
        return t
    starts, _ends = result
    before = [s for s in starts if s <= t]
    after = [s for s in starts if s > t]
    preferred, fallback = (before, after) if safe == "before" else (after, before)
    for candidates in (preferred, fallback):
        if candidates:
            return min(candidates, key=lambda start: abs(start - t))
    return t
```

```python
_XCORR_NFFT = 1024

_XCORR_PROBE = 2.5               # seconds inside the shared region where similarity must already be high
_XCORR_SNAP_WINDOW = 1.0         # seconds; refined boundaries only snap to a silence this close, safe side only
```

```python
def _log_band_spectrogram(pcm: np.ndarray) -> Optional[np.ndarray]:
    """40 log-spaced bands × 25 ms frames, each frame z-normalised, so a frame-wise dot product is a cosine similarity."""
    if pcm is None or pcm.size < _XCORR_NFFT:
        return None
    frames = np.lib.stride_tricks.sliding_window_view(pcm, _XCORR_NFFT)[::_XCORR_HOP] * np.hanning(_XCORR_NFFT)
    mag = np.abs(np.fft.rfft(frames, axis=1)).T  # (513, T)
    edges = np.unique(np.geomspace(4, 512, 41).astype(int))
    bands = np.stack([mag[edges[i]:edges[i + 1]].mean(axis=0) for i in range(len(edges) - 1)])
    spec = np.log1p(bands * 100.0)
    spec -= spec.mean(axis=0, keepdims=True)
    spec /= spec.std(axis=0, keepdims=True) + 1e-6
    return spec


def _xcorr_boundary(ep_path: str, t_ep: float, ref_path: str, t_ref: float, shared: str) -> Optional[float]:
    """Move *t_ep* to where the audio shared with *ref_path* really starts or ends.

    Both files are decoded from ``t - pre`` to ``t + _XCORR_WINDOW`` (same *pre* so the raw
    boundaries line up), the best lag within ±_XCORR_MAX_LAG aligns them, and the boundary is the
    edge of the run of frames whose similarity exceeds _XCORR_SIM_THRESHOLD, walked outward from a
    probe point _XCORR_PROBE s inside the shared side.  *shared* is ``"before"`` when the shared
    audio ends at the boundary (intro_end) and ``"after"`` when it begins there (intro_start,
    outro_start).  Returns ``None`` when audio cannot be decoded or the probe point is not similar.
    """
    fps = _XCORR_SR / _XCORR_HOP
    pre = min(_XCORR_WINDOW, t_ep, t_ref)
    ref_spec = _log_band_spectrogram(_extract_audio_pcm(ref_path, t_ref - pre, pre + _XCORR_WINDOW, sr=_XCORR_SR))
    ep_spec = _log_band_spectrogram(_extract_audio_pcm(ep_path, t_ep - pre, pre + _XCORR_WINDOW, sr=_XCORR_SR))
    if ref_spec is None or ep_spec is None:
        return None
    n = min(ref_spec.shape[1], ep_spec.shape[1])
    ref_spec, ep_spec = ref_spec[:, :n], ep_spec[:, :n]

    max_lag = int(_XCORR_MAX_LAG * fps)
    best_lag, best_score = 0, -np.inf
    for lag in range(-max_lag, max_lag + 1):  # ep[t] ~ ref[t + lag]
        a0, b0 = max(0, lag), max(0, -lag)
        m = n - abs(lag)
        if m <= 0:
            continue
        score = float((ref_spec[:, a0:a0 + m] * ep_spec[:, b0:b0 + m]).mean())
        if score > best_score:
            best_lag, best_score = lag, score
    a0, b0 = max(0, best_lag), max(0, -best_lag)
    m = n - abs(best_lag)
    sim = (ref_spec[:, a0:a0 + m] * ep_spec[:, b0:b0 + m]).mean(axis=0)
    smooth = np.convolve(sim, np.ones(_XCORR_SMOOTH) / _XCORR_SMOOTH, mode="same")
    good = smooth > _XCORR_SIM_THRESHOLD

    centre = int(round(pre * fps)) - b0  # frame of t_ep in the aligned ep spectrogram
    probe = int(_XCORR_PROBE * fps)
    if shared == "before":
        i = centre - probe
        if i < 0 or i >= m or not good[i]:
            return None
        while i + 1 < m and good[i + 1]:
            i += 1
        edge = i + 1
    else:
        i = centre + probe
        if i < 0 or i >= m or not good[i]:
            return None
        while i - 1 >= 0 and good[i - 1]:
            i -= 1
        edge = i
    return (t_ep - pre) + (edge + b0) * _XCORR_HOP / _XCORR_SR
```
